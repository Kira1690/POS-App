### 2.4 Calculation Logic

```typescript
const calculateEqualSplit = (bill: Bill, guestCount: number): GuestSplit[] => {
  const baseAmount = Math.floor((bill.totalAmount / guestCount) * 100) / 100;
  const remainder = bill.totalAmount - (baseAmount * guestCount);

  return Array.from({ length: guestCount }, (_, i) => ({
    id: generateId('GST'),
    guestNumber: i + 1,
    guestName: `Guest ${i + 1}`,
    items: [], // No specific items for equal split
    sharedItems: [],
    subtotal: bill.subtotal / guestCount,
    taxAmount: bill.taxAmount / guestCount,
    tipAmount: bill.tipAmount / guestCount,
    serviceCharge: bill.serviceCharge / guestCount,
    totalAmount: i === guestCount - 1 ? baseAmount + remainder : baseAmount,
    paymentStatus: 'unpaid',
    paidAmount: 0,
  }));
};
```

### 4.4 Validation Logic

```typescript
const validatePayments = (): { isValid: boolean; errors: string[] } => {
  const errors: string[] = [];
  const totalPayments = payments.reduce((sum, p) => sum + p.amount, 0);

  if (totalPayments < bill.totalAmount) {
    errors.push(`Payments total $${totalPayments.toFixed(2)} but $${bill.totalAmount.toFixed(2)} is due`);
  }

  if (totalPayments > bill.totalAmount) {
    errors.push(`Payments exceed total by $${(totalPayments - bill.totalAmount).toFixed(2)}`);
  }

  for (const payment of payments) {
    if (payment.amount <= 0) {
      errors.push(`Payment ${payment.id} must be greater than $0`);
    }
  }

  return { isValid: errors.length === 0, errors };
};
```

### 5.4 Validation Logic

```typescript
const validateComboSelection = (): { isValid: boolean; errors: string[] } => {
  const errors: string[] = [];

  for (const component of combo.components) {
    const selected = selections[component.id] || [];
    const totalQuantity = selected.reduce((sum, s) => sum + s.quantity, 0);

    if (totalQuantity < component.requiredQuantity) {
      errors.push(`Please select ${component.requiredQuantity} ${component.name}`);
    }
  }

  return { isValid: errors.length === 0, errors };
};
```
